# Open Measure: Complete Build Instructions for Claude Code

## Project Overview

Build **Open Measure**, an open-source measurement/analytics library for web applications. The library provides a unified API for tracking events across multiple analytics platforms (GA4, Meta Pixel, PostHog) with built-in consent management, deduplication, and GTM support.

**Primary Goal:** A library that developers (and AI coding tools) can install via npm and configure in minutes.

```bash
npm install @open-measure/web
```

```typescript
import { createTracker } from '@open-measure/web'

const tracker = createTracker({
  destinations: ['ga4', 'meta', 'posthog'],
  ga4: { measurementId: 'G-XXXXX' },
  meta: { pixelId: '123456' },
  posthog: { apiKey: 'phc_xxxxx' }
})

tracker.page()
tracker.track('purchase', { transaction_id: 'order_123', value: 99.99, currency: 'USD', items: [...] })
```

## Critical Requirements (From Original Plan)

### 1. Event Duplication Prevention
- event_id (UUID) required for all events
- Dedupe configurable per event type via unique_by field
- Dedupe window configurable (default 5 minutes)
- Storage options: memory, sessionStorage, localStorage
- GTM mode prevents double-firing

### 2. Consent/Privacy/PII
- consent() as first-class API
- Consent categories: necessary, analytics, marketing, personalization
- PII detection with console warnings for unhashed data
- Server-side hashing in CAPI module

### 3. GTM Integration
- Two modes:
  - mirror: Push to dataLayer AND fire destinations
  - authoritative: Push to dataLayer ONLY (GTM handles all tags)
- DataLayer format: { event: 'om_<event>', om_event_id, om_params, ... }

### 4. Platform Robustness
- Handle adblock/ITP via optional CAPI
- Graceful degradation when scripts fail
- Error boundaries with silent fail + console warnings

### 5. LLM-Friendly
- JSON spec as source of truth
- Auto-generated llm_context.md (<4000 tokens)
- MEASUREMENT_REPORT.md generated by CLI

### src/tracker.ts
```typescript
import { v4 as uuidv4 } from 'uuid'
import type { TrackerConfig, Tracker, TrackedEvent, ConsentState } from './types'
import { createDedupeEngine } from './dedupe'
import { createLogger } from './logger'
import { createDestination } from './destinations'

const DEFAULT_CONSENT: ConsentState = {
  necessary: true, analytics: false, marketing: false, personalization: false
}

const DEFAULT_DEDUPE = { enabled: true, window: 300000, storage: 'memory' as const }

export function createTracker(config: TrackerConfig): Tracker {
  let consentState = { ...DEFAULT_CONSENT, ...config.consent }
  const dedupeConfig = { ...DEFAULT_DEDUPE, ...config.dedupe }
  
  const logger = createLogger(config.debug ?? false)
  const dedupe = createDedupeEngine(dedupeConfig)
  
  const destinations = config.destinations
    .map(type => createDestination(type, config[type as keyof TrackerConfig], logger))
    .filter(Boolean)
  
  const anonymousId = getOrCreateAnonymousId()
  
  function processEvent(event: TrackedEvent): void {
    if (dedupe.isDuplicate(event)) {
      logger.debug('Event deduplicated', event.event_name)
      return
    }
    dedupe.markSeen(event)
    
    for (const dest of destinations) {
      if (dest.shouldTrack(event, consentState)) {
        try { dest.track(event) }
        catch (e) { logger.error(`Failed: ${dest.name}`, e) }
      }
    }
  }
  
  return {
    track(eventName, params = {}) {
      processEvent({
        event_name: eventName,
        event_id: uuidv4(),
        timestamp: new Date().toISOString(),
        anonymous_id: anonymousId,
        consent: { ...consentState },
        params
      })
    },
    page(params = {}) {
      this.track('page_view', {
        page_title: typeof document !== 'undefined' ? document.title : undefined,
        page_location: typeof window !== 'undefined' ? window.location.href : undefined,
        ...params
      })
    },
    identify(userId, traits = {}) {
      for (const dest of destinations) {
        dest.identify?.(userId, traits)
      }
    },
    consent(state) {
      consentState = { ...consentState, ...state }
      for (const dest of destinations) dest.onConsentChange?.(consentState)
    },
    getConsent: () => ({ ...consentState }),
    debug: (enabled) => logger.setDebug(enabled)
  }
}

function getOrCreateAnonymousId(): string {
  if (typeof window === 'undefined') return uuidv4()
  const key = 'om_anonymous_id'
  let id = localStorage.getItem(key)
  if (!id) { id = uuidv4(); try { localStorage.setItem(key, id) } catch {} }
  return id
}
```


### src/dedupe.ts
```typescript
import type { TrackedEvent } from './types'

const UNIQUE_BY: Record<string, string[]> = {
  purchase: ['transaction_id'],
  sign_up: ['user_id', 'method'],
  generate_lead: ['lead_id'],
  add_to_cart: ['event_id'],
  page_view: ['page_location', 'timestamp_bucket']
}

export function createDedupeEngine(config: { enabled: boolean; window: number; storage: string }) {
  const cache = new Map<string, number>()
  
  function getKey(event: TrackedEvent): string {
    const fields = UNIQUE_BY[event.event_name] || ['event_id']
    return [event.event_name, ...fields.map(f => 
      f === 'event_id' ? event.event_id :
      f === 'timestamp_bucket' ? Math.floor(Date.now() / 1000) :
      event.params[f] ?? event[f as keyof TrackedEvent]
    )].join(':')
  }
  
  return {
    isDuplicate(event: TrackedEvent): boolean {
      if (!config.enabled) return false
      const key = getKey(event)
      const seen = cache.get(key)
      return !!(seen && Date.now() - seen < config.window)
    },
    markSeen(event: TrackedEvent): void {
      if (config.enabled) cache.set(getKey(event), Date.now())
    },
    getCache: () => new Map(cache),
    clear: () => cache.clear()
  }
}
```

### src/destinations/ga4.ts
```typescript
import type { TrackedEvent, GA4Config, ConsentState } from '../types'

declare global { interface Window { dataLayer: unknown[]; gtag: (...args: unknown[]) => void } }

export class GA4Destination {
  name = 'ga4'
  consentCategory: keyof ConsentState = 'analytics'
  private config: GA4Config
  private initialized = false
  
  constructor(config: GA4Config, private logger: any) { this.config = config }
  
  async init() {
    if (typeof window === 'undefined') return
    window.dataLayer = window.dataLayer || []
    window.gtag = function() { window.dataLayer.push(arguments) }
    window.gtag('js', new Date())
    window.gtag('config', this.config.measurementId, { send_page_view: false })
    await this.loadScript(`https://www.googletagmanager.com/gtag/js?id=${this.config.measurementId}`)
    this.initialized = true
  }
  
  track(event: TrackedEvent) {
    if (!this.initialized) return
    window.gtag('event', event.event_name, { ...event.params, om_event_id: event.event_id })
  }
  
  identify(userId: string, traits?: Record<string, unknown>) {
    if (this.initialized) window.gtag('config', this.config.measurementId, { user_id: userId, user_properties: traits })
  }
  
  shouldTrack(_: TrackedEvent, consent: ConsentState) { return consent.analytics }
  onConsentChange?() {}
  
  private loadScript(src: string): Promise<void> {
    return new Promise((res, rej) => {
      if (typeof document === 'undefined') return res()
      const s = document.createElement('script')
      s.src = src; s.async = true
      s.onload = () => res(); s.onerror = () => rej()
      document.head.appendChild(s)
    })
  }
}
```

### src/destinations/meta.ts
```typescript
import type { TrackedEvent, MetaConfig, ConsentState, Item } from '../types'

declare global { interface Window { fbq: any; _fbq: any } }

const MAP: Record<string, string> = {
  page_view: 'PageView', sign_up: 'CompleteRegistration', generate_lead: 'Lead',
  add_to_cart: 'AddToCart', begin_checkout: 'InitiateCheckout', purchase: 'Purchase'
}

export class MetaDestination {
  name = 'meta'
  consentCategory: keyof ConsentState = 'marketing'
  private config: MetaConfig
  private initialized = false
  
  constructor(config: MetaConfig, private logger: any) { this.config = config }
  
  async init() {
    if (typeof window === 'undefined') return
    const fbq = window.fbq = window.fbq || function() { fbq.callMethod ? fbq.callMethod.apply(fbq, arguments) : fbq.queue.push(arguments) }
    if (!window._fbq) window._fbq = fbq
    fbq.queue = fbq.queue || []; fbq.loaded = true; fbq.version = '2.0'
    await this.loadScript('https://connect.facebook.net/en_US/fbevents.js')
    window.fbq('init', this.config.pixelId)
    this.initialized = true
  }
  
  track(event: TrackedEvent) {
    if (!this.initialized) return
    this.checkPii(event.params)
    const metaEvent = MAP[event.event_name] || event.event_name
    const params = this.mapParams(event)
    window.fbq('track', metaEvent, params, { eventID: event.event_id })
  }
  
  identify(userId: string, traits?: Record<string, unknown>) {
    if (!this.initialized || !traits) return
    const am: Record<string, string> = {}
    if (traits.email) { if (!this.isHashed(traits.email as string)) this.logger.warn('Meta: hash email'); am.em = traits.email as string }
    if (Object.keys(am).length) window.fbq('init', this.config.pixelId, am)
  }
  
  shouldTrack(_: TrackedEvent, consent: ConsentState) { return consent.marketing }
  
  private mapParams(e: TrackedEvent) {
    const p: Record<string, unknown> = {}
    if (e.params.value !== undefined) p.value = e.params.value
    if (e.params.currency) p.currency = e.params.currency
    if (e.params.transaction_id) p.order_id = e.params.transaction_id
    if (Array.isArray(e.params.items)) {
      p.content_ids = (e.params.items as Item[]).map(i => i.item_id)
      p.content_type = 'product'
    }
    return p
  }
  
  private checkPii(params: Record<string, unknown>) {
    ['email', 'phone'].forEach(f => {
      if (params[f] && typeof params[f] === 'string' && !this.isHashed(params[f] as string))
        this.logger.warn(`Meta: "${f}" contains unhashed PII`)
    })
  }
  
  private isHashed(v: string) { return /^[a-f0-9]{64}$/i.test(v) }
  
  private loadScript(src: string): Promise<void> {
    return new Promise((res) => {
      if (typeof document === 'undefined') return res()
      const s = document.createElement('script')
      s.src = src; s.async = true; s.onload = () => res()
      document.head.appendChild(s)
    })
  }
}
```

### src/adapter.ts
```typescript
import type { TrackedEvent, GTMConfig, ConsentState } from '@open-measure/web'

declare global { interface Window { dataLayer: Record<string, unknown>[] } }

export class GTMDestination {
  name = 'gtm'
  private config: GTMConfig
  private initialized = false
  
  constructor(config: GTMConfig) {
    this.config = { mode: 'mirror', dataLayerName: 'dataLayer', ...config }
  }
  
  async init() {
    if (typeof window === 'undefined') return
    window.dataLayer = window.dataLayer || []
    this.initialized = true
    if (this.config.mode === 'mirror') {
      console.warn('[OpenMeasure] GTM in "mirror" mode - events sent to both dataLayer AND destinations')
    }
  }
  
  track(event: TrackedEvent) {
    if (!this.initialized) return
    window.dataLayer.push({
      event: `om_${event.event_name}`,
      om_event_id: event.event_id,
      om_timestamp: event.timestamp,
      om_params: event.params,
      ...this.flatten(event.params)
    })
  }
  
  identify(userId: string, traits?: Record<string, unknown>) {
    if (this.initialized) window.dataLayer.push({ event: 'om_identify', om_user_id: userId, om_user_traits: traits })
  }
  
  onConsentChange(consent: ConsentState) {
    if (this.initialized) window.dataLayer.push({ event: 'om_consent_update', om_consent: consent })
  }
  
  shouldTrack() { return true }
  
  private flatten(p: Record<string, unknown>) {
    const f: Record<string, unknown> = {}
    for (const [k, v] of Object.entries(p)) if (typeof v !== 'object') f[`om_${k}`] = v
    return f
  }
}
```

```typescript
'use client'
import React, { createContext, useContext, useRef, useMemo } from 'react'
import { createTracker, type Tracker, type TrackerConfig } from '@open-measure/web'

const Ctx = createContext<{ tracker: Tracker | null }>({ tracker: null })

export function OpenMeasureProvider({ config, children }: { config: TrackerConfig; children: React.ReactNode }) {
  const ref = useRef<Tracker | null>(null)
  if (!ref.current) ref.current = createTracker(config)
  const value = useMemo(() => ({ tracker: ref.current }), [])
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>
}

export function useOpenMeasure(): Tracker {
  const { tracker } = useContext(Ctx)
  if (!tracker) throw new Error('useOpenMeasure must be within OpenMeasureProvider')
  return tracker
}
```

### src/invariants.ts
```typescript
import type { TrackedEvent } from '@open-measure/web'

export interface Invariant {
  name: string
  event: string | '*'
  condition: (e: TrackedEvent) => boolean
  message: string
  severity: 'error' | 'warning'
}

export const BUILTIN_INVARIANTS: Invariant[] = [
  { name: 'purchase_transaction_id', event: 'purchase', condition: e => typeof e.params.transaction_id === 'string' && e.params.transaction_id.length > 0, message: 'purchase needs transaction_id', severity: 'error' },
  { name: 'purchase_value', event: 'purchase', condition: e => typeof e.params.value === 'number' && e.params.value > 0, message: 'purchase needs positive value', severity: 'error' },
  { name: 'purchase_currency', event: 'purchase', condition: e => typeof e.params.currency === 'string' && /^[A-Z]{3}$/.test(e.params.currency), message: 'purchase needs ISO currency', severity: 'error' },
  { name: 'purchase_items', event: 'purchase', condition: e => Array.isArray(e.params.items) && e.params.items.length > 0, message: 'purchase needs items', severity: 'error' },
  { name: 'sign_up_method', event: 'sign_up', condition: e => typeof e.params.method === 'string', message: 'sign_up should have method', severity: 'warning' }
]

export function createInvariantEngine(custom: Invariant[] = []) {
  const all = [...BUILTIN_INVARIANTS, ...custom]
  return {
    validate: (e: TrackedEvent) => all.filter(i => i.event === '*' || i.event === e.event_name).map(i => ({ invariant: i, passed: i.condition(e) })),
    add: (i: Invariant) => all.push(i),
    list: () => [...all]
  }
}
```

```typescript
    const files = await glob(['**/*.{ts,tsx}'], { cwd: opts.src, ignore: ['node_modules/**'] })
    const results: { file: string; event: string; issues: string[] }[] = []
    
    for (const file of files) {
      const content = fs.readFileSync(path.join(opts.src, file), 'utf-8')
      const matches = [...content.matchAll(/\.track\(['"](\w+)['"]/g)]
      for (const m of matches) {
        const event = m[1]
        const issues: string[] = []
        if (event === 'purchase' && !content.includes('transaction_id')) issues.push('Missing transaction_id')
        results.push({ file, event, issues })
      }
    }
    
    let report = '# Measurement Report\n\n'
    report += `Generated: ${new Date().toISOString()}\n\n`
    report += `Total: ${results.length} tracking calls\n\n`
    for (const r of results) {
      report += `- ${r.file}: ${r.event}${r.issues.length ? ` ⚠️ ${r.issues.join(', ')}` : ''}\n`
    }
    
    fs.writeFileSync(opts.output, report)
    console.log(chalk.green(`✓ Report: ${opts.output}`))
  })
```

```typescript
import { createHash } from 'crypto'

export const hashPii = (v: string) => createHash('sha256').update(v.toLowerCase().trim()).digest('hex')
export const hashEmail = hashPii
export const hashPhone = (p: string) => hashPii(p.replace(/\D/g, ''))

export function hashUserData(d: Record<string, unknown>) {
  const r = { ...d }
  for (const f of ['email', 'phone', 'em', 'ph']) {
    if (typeof r[f] === 'string') r[f] = hashPii(r[f] as string)
  }
  return r
}
```

### src/next/route.ts
```typescript
import { NextRequest, NextResponse } from 'next/server'
import { hashUserData } from '../hashing'

export function createCAPIHandler(config: { meta?: { pixelId: string; accessToken: string }; hashPii?: boolean }) {
  return async (req: NextRequest) => {
    const { event_name, event_id, params, user_data } = await req.json()
    const userData = config.hashPii && user_data ? hashUserData(user_data) : user_data
    
    if (config.meta) {
      await fetch(`https://graph.facebook.com/v18.0/${config.meta.pixelId}/events?access_token=${config.meta.accessToken}`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ data: [{ event_name, event_id, event_time: Math.floor(Date.now() / 1000), action_source: 'website', user_data: userData, custom_data: params }] })
      })
    }
    
    return NextResponse.json({ success: true })
  }
}
```
